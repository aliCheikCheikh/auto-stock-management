package "Aggregate StorageLocation" as pkg_location <<Rectangle>> #FAECE7 {

  class StorageLocation <<Aggregate Root>> {
    - locationId : LocationId
    - shopId : ShopId
    - type : LocationType
    - label : String
    - lowStockIndicator : int
    - stockLevels : Map<ProductId, StockLevel>
    --
    + increaseStock(productId, qty) : void
    + decreaseStock(productId, qty) : void
    + getStockLevel(productId) : int
    + hasEnoughStock(productId, qty) : boolean
    + isShopFloorLow(productId) : boolean

    - productId : ProductId
    - quantity : int
    --
    + {static} of(productId, qty) : StockLevel
    + increase(amount : int) : StockLevel
    + decrease(amount : int) : StockLevel
  }

  StorageLocation "1" *-down- "0..*" StockLevel

  }

  }

    - value : UUID

SM ..> U : performedBy
SM ..> SA : saleId

SA ..> P : productId\n(via SaleLineItem)

note bottom of SM
  Central audit record.
  References most
  other aggregates.

note bottom of SL
  Protects the invariant
  "quantity ≥ 0"
